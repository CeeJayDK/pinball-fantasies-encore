# Names for places in TABLE1.PRG (the supported release), as they are worked out.
# "c" lines name code in segment 0010, "d" lines name data in segment 19b5.
# python3 re/disasm.py puts them into re/fantasy/TABLE1_seg0010.lst.

# --- the frame
c 2f9b entry
c 35fd mainLoop                 ; reads keys, counts loops, polls the driver
c 41c6 frameCallback            ; driver callback at the start of the frame
c 55db midFrameCallback         ; driver callback part way down the frame
c 3e44 keyboardInterrupt
c 87c0 physicsOnce              ; guard against re-entry round physicsSteps
c 87db physicsSteps             ; two sub-steps, or one in the mid-frame callback in high resolution
c 8829 probeBall                ; carry clear when the ball touches something
c 8e95 bounce
c 9133 nudgeAndFlippers
c 908f integrate
c 9106 stampFlippers
c 59aa afterSteps               ; tilt decay, bumper events, gravity, layer
c 59d9 pickGravity
c 5d24 changeLayer
c 5989 runRules                 ; roll triggers, hit triggers, the table's own per-frame code, timers
c 5d70 rollTriggers
c 5cb9 hitTriggers
c 5ace bumperEvent
c 4018 scroll
